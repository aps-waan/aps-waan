def auto_detect_types(df: pd.DataFrame) -> dict:
    profile = {}
    for col in df.columns:
        if pd.api.types.is_numeric_dtype(df[col]):
            profile[col] = "METRIC"
        elif pd.api.types.is_datetime64_any_dtype(df[col]):
            profile[col] = "DATETIME"
        else:
            profile[col] = "DIMENSION"
    return profile
